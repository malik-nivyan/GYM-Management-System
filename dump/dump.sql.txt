-- dump.sql
-- Schema and sample data for the GYM management app (database: `dump`)
-- Created to match usage seen in GYM_Management_App.java

DROP DATABASE IF EXISTS `dump`;
CREATE DATABASE `dump` CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE `dump`;

-- membership table
CREATE TABLE `membership` (
  `Membership_id` INT NOT NULL,
  `Name` VARCHAR(100) NOT NULL,
  `Duration_months` INT NOT NULL,
  `Fee` DECIMAL(10,2) NOT NULL,
  PRIMARY KEY (`Membership_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- trainer table
CREATE TABLE `trainer` (
  `Trainer_id` INT NOT NULL,
  `First_name` VARCHAR(100),
  `Last_name` VARCHAR(100),
  `Gender` VARCHAR(10),
  `Age` INT,
  `Phone` VARCHAR(30),
  `Email` VARCHAR(150),
  `Specialization` VARCHAR(150),
  `Hire_date` DATE,
  PRIMARY KEY (`Trainer_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- member table
CREATE TABLE `member` (
  `Member_id` INT NOT NULL,
  `First_name` VARCHAR(100),
  `Last_name` VARCHAR(100),
  `Gender` VARCHAR(10),
  `Age` INT,
  `Phone` VARCHAR(30),
  `Email` VARCHAR(150),
  `Join_date` DATE,
  `Trainer_id` INT,
  `Status` VARCHAR(20) DEFAULT 'Active',
  `Membership_id` INT,
  PRIMARY KEY (`Member_id`),
  CONSTRAINT `fk_member_trainer` FOREIGN KEY (`Trainer_id`) REFERENCES `trainer`(`Trainer_id`)
    ON DELETE RESTRICT ON UPDATE CASCADE,
  CONSTRAINT `fk_member_membership` FOREIGN KEY (`Membership_id`) REFERENCES `membership`(`Membership_id`)
    ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- payment table
CREATE TABLE `payment` (
  `Payment_id` INT NOT NULL,
  `Member_id` INT NOT NULL,
  `Amount` DECIMAL(10,2) NOT NULL,
  `Payment_date` DATE,
  `Payment_method` VARCHAR(20),
  `Status` VARCHAR(20),
  PRIMARY KEY (`Payment_id`),
  CONSTRAINT `fk_payment_member` FOREIGN KEY (`Member_id`) REFERENCES `member`(`Member_id`)
    ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- login table (used for authenticateAndGetRole)
CREATE TABLE `login` (
  `Username` VARCHAR(100) NOT NULL,
  `Password` VARCHAR(255) NOT NULL,
  `Role` VARCHAR(50) DEFAULT 'Admin',
  PRIMARY KEY (`Username`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Indexes to speed up searches used by the app
CREATE INDEX `idx_member_firstname` ON `member` (`First_name`);
CREATE INDEX `idx_member_lastname`  ON `member` (`Last_name`);
CREATE INDEX `idx_member_email`     ON `member` (`Email`);
CREATE INDEX `idx_payment_status`   ON `payment` (`Status`);
